cmake_minimum_required(VERSION 3.20)
project(forma_dynamictext LANGUAGES C CXX)

# A WebAssembly parent supplies pinned, Emscripten-built freetype and harfbuzz targets.
if(NOT TARGET freetype OR NOT TARGET harfbuzz)
  if(EMSCRIPTEN)
    message(FATAL_ERROR "Add pinned FreeType and HarfBuzz source targets before Forma dynamictext; host libraries cannot link into WASM.")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FT REQUIRED IMPORTED_TARGET freetype2)
  pkg_check_modules(HB REQUIRED IMPORTED_TARGET harfbuzz)
  set(FORMA_TEXT_LIBRARIES PkgConfig::FT PkgConfig::HB)
else()
  set(FORMA_TEXT_LIBRARIES freetype harfbuzz)
endif()

if(EMSCRIPTEN)
  add_library(forma_dynamictext STATIC forma_dynamictext.c)
else()
  add_library(forma_dynamictext forma_dynamictext.c)
endif()
target_link_libraries(forma_dynamictext PUBLIC ${FORMA_TEXT_LIBRARIES})
target_include_directories(forma_dynamictext PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(forma_dynamictext PRIVATE c_std_11)
set_target_properties(forma_dynamictext PROPERTIES POSITION_INDEPENDENT_CODE ON)
